-- YARG Phase 4: session-scoped, shared, fair performance rotation. -- Run after the Phase 1/2/3 migrations. Read-only; does NOT reorder or alter stored requests. -- Designed to be reused by the future admin.html MC dashboard. BEGIN; -- Canonical rockstar key: one anonymous session may represent several individually named people. -- If a player uses another browser/login, the system cannot prove they are the same person. CREATE OR REPLACE FUNCTION public.yarg_fair_participant_keys(p_request_id uuid) RETURNS TABLE(participant_key text) LANGUAGE sql STABLE SECURITY DEFINER SET search_path = '' AS $$ SELECT DISTINCT person_key FROM ( SELECT ('guest:' || COALESCE(m.guest_user_id::text, 'unknown') || ':' || lower(btrim(m.rockstar_name))) AS person_key FROM public.performance_members m WHERE m.performance_request_id = p_request_id AND m.withdrawn_at IS NULL AND nullif(btrim(m.rockstar_name), '') IS NOT NULL UNION ALL SELECT ('guest:' || COALESCE(r.guest_user_id::text, 'unknown') || ':' || lower(btrim(r.rockstar_name))) AS person_key FROM public.performance_requests r WHERE r.id = p_request_id AND coalesce(r.is_group_request, false) = false AND r.requester_withdrawn_at IS NULL AND nullif(btrim(r.rockstar_name), '') IS NOT NULL ) people; $$; -- Returns ONLY publicly visible request IDs and their shared queue positions. -- The playing song is anchored at the top. Completed turns are counted, then each -- upcoming band's participants receive one virtual turn as that song is scheduled. -- Max participant turns is the main score: a veteran cannot jump ahead by joining -- a first-timer's band. Sum of turns is the tiebreaker; FIFO resolves remaining ties. CREATE OR REPLACE FUNCTION public.yarg_get_fair_queue() RETURNS TABLE(request_id uuid, fair_position integer, highest_turns integer) LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path = '' AS $$ DECLARE v_session uuid; v_counts jsonb := '{}'::jsonb; v_used uuid[] := '{}'::uuid[]; v_row record; v_other record; v_key record; v_pos integer := 0; v_max integer; v_sum integer; v_best_id uuid; v_best_max integer; v_best_sum integer; v_best_created timestamptz; v_best_turns integer; v_turns integer; BEGIN SELECT ps.current_session_id INTO v_session FROM public.party_settings ps WHERE ps.id=1; IF v_session IS NULL THEN RETURN; END IF; -- Count each completed group performance once for each active participating rockstar. FOR v_row IN SELECT r.id FROM public.performance_requests r WHERE r.party_session_id=v_session AND r.status IN ('completed','played') LOOP FOR v_key IN SELECT participant_key FROM public.yarg_fair_participant_keys(v_row.id) LOOP v_counts := jsonb_set(v_counts, ARRAY[v_key.participant_key], to_jsonb(COALESCE((v_counts->>v_key.participant_key)::integer,0)+1), true); END LOOP; END LOOP; -- The currently playing performance always comes first, not re-ranked. FOR v_row IN SELECT r.id FROM public.performance_requests r WHERE r.party_session_id=v_session AND r.status='playing' ORDER BY r.started_at NULLS LAST,r.created_at,r.id LOOP v_pos:=v_pos+1; v_used:=array_append(v_used,v_row.id); request_id:=v_row.id; fair_position:=v_pos; highest_turns:=NULL; RETURN NEXT; -- Reserve turns for the live performance so its members don't go again immediately. FOR v_key IN SELECT participant_key FROM public.yarg_fair_participant_keys(v_row.id) LOOP v_counts := jsonb_set(v_counts, ARRAY[v_key.participant_key], to_jsonb(COALESCE((v_counts->>v_key.participant_key)::integer,0)+1), true); END LOOP; END LOOP; -- Small party-sized queues: greedy, stable recomputation after every selected slot. LOOP v_best_id:=NULL; v_best_max:=NULL; v_best_sum:=NULL; v_best_created:=NULL; FOR v_other IN SELECT r.id,r.created_at FROM public.performance_requests r WHERE r.party_session_id=v_session AND r.status IN ('waiting','full') AND NOT (r.id=ANY(v_used)) ORDER BY r.created_at,r.id LOOP v_max:=0; v_sum:=0; FOR v_key IN SELECT participant_key FROM public.yarg_fair_participant_keys(v_other.id) LOOP v_turns:=COALESCE((v_counts->>v_key.participant_key)::integer,0); v_max:=GREATEST(v_max,v_turns); v_sum:=v_sum+v_turns; END LOOP; IF v_best_id IS NULL OR v_max>v_key.participant_key)::integer,0)+1), true); END LOOP; END LOOP; END; $$; REVOKE ALL ON FUNCTION public.yarg_fair_participant_keys(uuid) FROM PUBLIC,anon,authenticated; REVOKE ALL ON FUNCTION public.yarg_get_fair_queue() FROM PUBLIC; -- Both signed-in guests and non-signed-in viewers can see the same public IDs/order. GRANT EXECUTE ON FUNCTION public.yarg_get_fair_queue() TO anon,authenticated; COMMIT;
